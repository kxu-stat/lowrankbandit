% Animation of Algorithm 1 on its figure (greedy_flow_body.tex).

% Every overlay shows the whole figure (\input{greedy_flow_body}); the overlays only add, on top
% (animation_style.tex): a red marker on the horizon strip for the current round t, red frames and red
% copies of the arrows for the step being carried out, white veils over what is not used at that moment
% and over record cells not yet added, and the caption bar.
% Each overlay is one frame; build_animation.py sets how long each frame is shown.
%
%   overlay  1       overview
%   overlays 2-9     exploration: round 1 step by step, rounds 2 and tau, the first fit
%   overlays 10-17   greedy: round tau+1 step by step
%   overlays 18-21   later rounds, round n, the complete figure

\only<1>{\captionbar{chipgray}{Overview}{Algorithm 1, the policy that collects the data: a short
  exploration phase, then greedy choices from a refitted low-rank model.}}%
\only<2>{\captionbar{chipgray}{Explore}{Round \(t\le\tau\): observe the user and the candidate ads.}}%
\only<3>{\captionbar{chipgray}{Explore}{An ad is drawn at random; no fitted model is used.}}%
\only<4-5>{\captionbar{chipgray}{Explore}{The ad \(a_t\) is displayed, and only this ad gives a
  response: click or no click.}}%
\only<6>{\captionbar{chipgray}{Explore}{The round is added to the data.}}%
\only<7-8>{\captionbar{chipgray}{Explore}{Rounds \(2,\ldots,\tau\) do the same, each ad drawn
  independently of the past.}}%
\only<9>{\captionbar{chipgray}{Explore}{At \(t=\tau\), the first nuclear-norm-penalized fit is computed
  from the exploration data.}}%
\only<10>{\captionbar{chipteal}{Greedy}{Round \(t>\tau\): observe the user and the candidate ads.}}%
\only<11>{\captionbar{chipteal}{Greedy}{Score each ad with the latest fit \(\hat\theta_{t-1}\): ad
  effect, user effect and a low-rank user--ad interaction.}}%
\only<12>{\captionbar{chipteal}{Greedy}{Show the ad with the largest score. The choice is made before
  its response is seen.}}%
\only<13-14>{\captionbar{chipteal}{Greedy}{Display \(a_t\) and observe click or no click.}}%
\only<15-16>{\captionbar{chipteal}{Greedy}{Add the round to the data and refit the
  nuclear-norm-penalized model on all data so far.}}%
\only<17>{\captionbar{chipteal}{Greedy}{The new fit \(\hat\theta_t\) scores the ads in the next
  round.}}%
\only<18-19>{\captionbar{chipteal}{Greedy}{Every later round repeats the greedy loop; there is no
  further exploration.}}%
\only<20>{\captionbar{chipteal}{Greedy}{The loop runs until round \(n\).}}%
\only<21>{\captionbar{chipgray}{Overview}{The exploration length \(\tau\) balances the accuracy of the
  first fit against the clicks forgone while ads are drawn at random.}}%
